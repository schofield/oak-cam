# Probe: resolve the installed depthai-core package exactly as a consumer would
# (find_package(depthai CONFIG)) and dump the include directories, compile
# definitions, link libraries and the shared library location to oak-flags.json
# in the install prefix. crates/depthai-sys/build.rs reads that file instead of
# guessing where vcpkg put nlohmann/json, spdlog, libnop and friends.
cmake_minimum_required(VERSION 3.20)
project(oak_depthai_probe LANGUAGES CXX)

find_package(depthai CONFIG REQUIRED)

set(_tgt depthai::core)
if(NOT TARGET ${_tgt})
    message(FATAL_ERROR "depthai::core target not found in ${depthai_DIR}")
endif()

# Walk INTERFACE_LINK_LIBRARIES transitively collecting include dirs and
# definitions from every imported target we can see.
set(_seen "")
set(_includes "")
set(_defines "")
set(_libs "")
function(_collect tgt)
    if(NOT TARGET ${tgt})
        return()
    endif()
    list(FIND _seen ${tgt} _idx)
    if(NOT _idx EQUAL -1)
        return()
    endif()
    list(APPEND _seen ${tgt})
    set(_seen "${_seen}" PARENT_SCOPE)
    get_target_property(_inc ${tgt} INTERFACE_INCLUDE_DIRECTORIES)
    if(_inc)
        list(APPEND _includes ${_inc})
    endif()
    get_target_property(_sysinc ${tgt} INTERFACE_SYSTEM_INCLUDE_DIRECTORIES)
    if(_sysinc)
        list(APPEND _includes ${_sysinc})
    endif()
    get_target_property(_def ${tgt} INTERFACE_COMPILE_DEFINITIONS)
    if(_def)
        list(APPEND _defines ${_def})
    endif()
    get_target_property(_type ${tgt} TYPE)
    if(_type STREQUAL "SHARED_LIBRARY" OR _type STREQUAL "STATIC_LIBRARY" OR _type STREQUAL "UNKNOWN_LIBRARY")
        get_target_property(_loc ${tgt} IMPORTED_LOCATION_RELEASE)
        if(NOT _loc)
            get_target_property(_loc ${tgt} IMPORTED_LOCATION)
        endif()
        if(NOT _loc)
            get_target_property(_loc ${tgt} IMPORTED_LOCATION_NOCONFIG)
        endif()
        if(_loc)
            list(APPEND _libs "${_loc}")
        endif()
    endif()
    get_target_property(_deps ${tgt} INTERFACE_LINK_LIBRARIES)
    if(_deps)
        foreach(_d ${_deps})
            # Strip generator expressions like $<LINK_ONLY:foo>
            string(REGEX REPLACE "^\\$<LINK_ONLY:(.*)>$" "\\1" _d "${_d}")
            if(TARGET ${_d})
                _collect(${_d})
            endif()
        endforeach()
    endif()
    set(_includes "${_includes}" PARENT_SCOPE)
    set(_defines "${_defines}" PARENT_SCOPE)
    set(_libs "${_libs}" PARENT_SCOPE)
endfunction()
_collect(${_tgt})

# Drop generator expressions we cannot evaluate here (BUILD_INTERFACE etc).
set(_clean_inc "")
foreach(_i ${_includes})
    string(REGEX REPLACE "^\\$<INSTALL_INTERFACE:(.*)>$" "\\1" _i "${_i}")
    if(_i MATCHES "^\\$<")
        continue()
    endif()
    list(APPEND _clean_inc "${_i}")
endforeach()
list(REMOVE_DUPLICATES _clean_inc)
set(_clean_def "")
foreach(_d ${_defines})
    if(_d MATCHES "^\\$<")
        continue()
    endif()
    list(APPEND _clean_def "${_d}")
endforeach()
list(REMOVE_DUPLICATES _clean_def)
list(REMOVE_DUPLICATES _libs)

get_target_property(_core_loc ${_tgt} IMPORTED_LOCATION_RELEASE)
if(NOT _core_loc)
    get_target_property(_core_loc ${_tgt} IMPORTED_LOCATION)
endif()
if(NOT _core_loc)
    get_target_property(_core_loc ${_tgt} IMPORTED_LOCATION_NOCONFIG)
endif()

set(_json "{\n")
string(APPEND _json "  \"depthai_dir\": \"${depthai_DIR}\",\n")
string(APPEND _json "  \"version\": \"${depthai_VERSION}\",\n")
string(APPEND _json "  \"core_library\": \"${_core_loc}\",\n")
string(APPEND _json "  \"include_dirs\": [")
set(_first TRUE)
foreach(_i ${_clean_inc})
    if(NOT _first)
        string(APPEND _json ",")
    endif()
    string(APPEND _json "\n    \"${_i}\"")
    set(_first FALSE)
endforeach()
string(APPEND _json "\n  ],\n  \"defines\": [")
set(_first TRUE)
foreach(_d ${_clean_def})
    if(NOT _first)
        string(APPEND _json ",")
    endif()
    string(APPEND _json "\n    \"${_d}\"")
    set(_first FALSE)
endforeach()
string(APPEND _json "\n  ],\n  \"libraries\": [")
set(_first TRUE)
foreach(_l ${_libs})
    if(NOT _first)
        string(APPEND _json ",")
    endif()
    string(APPEND _json "\n    \"${_l}\"")
    set(_first FALSE)
endforeach()
string(APPEND _json "\n  ]\n}\n")

if(NOT DEFINED OAK_FLAGS_OUT)
    set(OAK_FLAGS_OUT "${CMAKE_INSTALL_PREFIX}/oak-flags.json")
endif()
file(WRITE "${OAK_FLAGS_OUT}" "${_json}")
message(STATUS "wrote ${OAK_FLAGS_OUT}")
